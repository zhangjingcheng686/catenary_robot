BMI088 是一款6轴惯性测量单元(\+IMU)，包含3轴加速度计和3轴陀螺仪。本驱动支持通过\+SPI接口与\+BMI088通信。\hypertarget{md_source_2bsp_2_b_m_i088_2_b_m_i088_autotoc_md2}{}\doxysection{\texorpdfstring{硬件连接}{硬件连接}}\label{md_source_2bsp_2_b_m_i088_2_b_m_i088_autotoc_md2}

\begin{DoxyItemize}
\item 使用\+SPI2接口 (hspi2)
\item 加速度计片选引脚\+: ACC\+\_\+\+CS\+\_\+\+Pin
\item 陀螺仪片选引脚\+: GYRO\+\_\+\+CS\+\_\+\+Pin
\item 需要\+TIM3用于延时(PWM模式)
\end{DoxyItemize}\hypertarget{md_source_2bsp_2_b_m_i088_2_b_m_i088_autotoc_md3}{}\doxysection{\texorpdfstring{初始化流程}{初始化流程}}\label{md_source_2bsp_2_b_m_i088_2_b_m_i088_autotoc_md3}

\begin{DoxyEnumerate}
\item 严格的时序要求：
\begin{DoxyItemize}
\item 寄存器写入后需要延迟 {\ttfamily BMI088\+\_\+\+COM\+\_\+\+WAIT\+\_\+\+SENSOR\+\_\+\+TIME} (us级)
\item 复位后需要延迟 {\ttfamily BMI088\+\_\+\+LONG\+\_\+\+DELAY\+\_\+\+TIME} (ms级)
\end{DoxyItemize}
\item 量程配置：
\begin{DoxyItemize}
\item 加速度计量程\+: 默认3G ({\ttfamily BMI088\+\_\+\+ACCEL\+\_\+3\+G\+\_\+\+SEN})
\item 陀螺仪量程\+: 默认2000dps ({\ttfamily BMI088\+\_\+\+GYRO\+\_\+2000\+\_\+\+SEN})
\end{DoxyItemize}
\item 错误处理：
\begin{DoxyItemize}
\item 初始化函数返回错误代码，需检查处理
\item 常见错误：
\begin{DoxyItemize}
\item {\ttfamily BMI088\+\_\+\+NO\+\_\+\+SENSOR}\+: 通信失败
\item {\ttfamily BMI088\+\_\+\+ACC\+\_\+\+PWR\+\_\+\+CTRL\+\_\+\+ERROR}\+: 加速度计电源控制错误
\item {\ttfamily BMI088\+\_\+\+GYRO\+\_\+\+RANGE\+\_\+\+ERROR}\+: 陀螺仪量程配置错误
\end{DoxyItemize}
\end{DoxyItemize}
\item 硬件依赖：
\begin{DoxyItemize}
\item 需要实现 {\ttfamily BMI088\+\_\+delay\+\_\+us()} 和 {\ttfamily BMI088\+\_\+delay\+\_\+ms()}
\item 需要配置正确的\+SPI接口和片选引脚 
\end{DoxyItemize}
\end{DoxyEnumerate}
